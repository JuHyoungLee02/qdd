\section*{큰 줄기}
\begin{itemize}
  \item \intent{기존 스킬 기반 잡기를 기본으로 쓰고, 실패할 때마다 GPT-6 Astra가 개입해서 이어감}
  \item \intent{상위 Astra(느림, 계획) / 하위 Jev(빠른 객관식 판단). 예전 VLA의 큰 LLM + 작은 VLM 구조처럼}
  \item \intent{VLA는 다른 환경으로 일반화가 안 됨. LLM은 됨. 항상 부각}
  \item \intent{안전은 크게 다루지 않음}
  \item \idea{계층 구조 자체는 이미 데모로 공개됨. 새로움은 방법(겹침 호출, 호출 시점, 스킬 결합)과 평가에서 찾음}
  \item \decide{핵심 주장 범위. 최신 VLA($\pi_{0.5}$ 등)가 새 환경을 주장하므로 ``대규모 로봇 데이터와 학습 없이 덜 무너짐''처럼 좁힐지}
\end{itemize}

\section*{모델 제약 (확인된 것만)}
\begin{itemize}
  \item Jev: 텍스트/JSON만. 수치·산술에 약함, 관련 없는 상태가 늘면 정확도 하락 $\rightarrow$ 기하는 코드에서 계산, Jev에는 범주·술어로
  \item Jev 확률 보정은 업체 주장뿐 $\rightarrow$ 우리 데이터로 먼저 측정
  \item Astra: 이미지는 받지만 비디오는 없음. 느림
  \item (v3에서 원문으로 재확인 중)
\end{itemize}

\section*{M1 상태 표현 (이미지 $\rightarrow$ Jev)}
\begin{itemize}
  \item \intent{Jev는 이미지를 못 봄. AI 전 분야에서 가장 효율적인 변환을 가져옴}
  \item \idea{작업 관련 물체만 + 코드가 계산한 범주형 관계 + 참/거짓 술어. 원좌표는 보조}
  \item \chg{v1의 ``원좌표 JSON'', ``변화분만 주기''는 뺌 (Jev 수치 약점, 근거 없음)}
  \item \decide{변환 방법과 인식 앞단은 사용자가 정함}
\end{itemize}

\section*{M2 Astra $\rightarrow$ Jev 전달}
\begin{itemize}
  \item \intent{M1과 거의 같은 문제. Astra는 이미지를 보지만 Jev는 못 봄}
  \item \idea{완료 조건 술어 + 대상 물체 + 예상 소요 시간 + 연결할 스킬}
\end{itemize}

\section*{M3 행동 표현}
\begin{itemize}
  \item \intent{촘촘하게 가도 됨. Jev 방식이면 가능}
  \item \idea{크기는 순서형 보기. 확률 평균은 한 봉우리일 때만}
  \item \dirn{스텝별 선택 vs 저주파 경유점 선택 비교. LLM 쪽 ``확률 $\rightarrow$ 연속값'' 방법 조사 중}
\end{itemize}

\section*{M4 연속 동작 1단계 (RTC식 겹침 호출)}
\begin{itemize}
  \item \intent{1초에 Jev를 여러 번(예: 3번) 계단식으로 겹쳐 호출}
  \item \intent{겹친 구간이 제대로 반영됐는지 스스로 확인하고 업데이트하는 시스템이 반드시 필요. 가장 중요}
  \item \idea{확정된 앞부분 고정 + 응답 도착 시점의 예상 상태를 입력 + 겹침 구간 융합 + 매 스텝 빠른 기하 보정}
  \item \dirn{가장 유력한 컨트리뷰션 자리. LLM 쪽 draft-verify 계열도 보는 중}
\end{itemize}

\section*{M5 연속 동작 2단계 (스무딩)}
\begin{itemize}
  \item \intent{아주 작은 움직임은 VLA식 스무딩으로}
  \item \idea{학습 없는 후처리(겹침 블렌딩 + jerk 최소화)}
\end{itemize}

\section*{M6 스킬 $\leftrightarrow$ Jev}
\begin{itemize}
  \item \intent{LLM이 스킬을 부르고 끝이 아니라 잘게 엮음. 1년 이내, 스타 상위 논문 참고}
  \item \idea{스킬 코드 안의 결정 지점을 Jev 질문으로 엶. 기반 후보 CaP-X}
  \item \dirn{스타 상위 5개 확정 중}
\end{itemize}

\section*{M7 진행 판별}
\begin{itemize}
  \item \intent{VLA에서 ``계획대로''를 어떻게 정의했는지 참고}
  \item \idea{완료는 센서·술어로, 진행도는 ``완료까지 남은 시간''으로. 접촉·파지는 VLM이 약하니 센서로}
\end{itemize}

\section*{M8 Astra 호출}
\begin{itemize}
  \item \intent{최소한 실패 판정 때 부름. 한 장이 아니라 연속 프레임(예: 한 이미지에 10개)}
  \item \intent{호출 방식 비교: 일정 주기 / 정한 순간 / ``이때쯤 나왔어야 하는데'' 자체 판단}
  \item \intent{처음 계획할 때만 멈춤. 이후엔 실패로 다시 생각할 때만}
  \item \chg{v2의 ``복구·주기 호출은 effort low''는 기본안에서 뺌. 사용자가 다른 작업에서 ``추론 강도를 낮추는 건 옳지 않다''고 함 $\rightarrow$ effort는 실험 축으로만}
  \item \dirn{세 호출 방식 비교가 이미 있는지 확인 중 (2608.28075)}
\end{itemize}
