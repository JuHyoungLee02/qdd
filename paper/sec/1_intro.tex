\section{Introduction --- 큰 줄기}
\label{sec:intro}

\begin{itemize}
  \item \intent{물체를 보고 스킬을 생성해서 잡는 방법들이 많지만 실패할 때가 있음. 실패할 때마다 GPT-6 Astra가 개입해서 진행하는 방식을 원함. 논문으로 제출할 계획}
  \item \intent{상위 Astra(느림, 계획) / 하위 Jev(빠른 객관식 판단). 예전 VLA의 큰 LLM + 작은 VLM처럼}
  \item \intent{Jev는 객관식 문제를 엄청 잘 풂. 엔드포인트를 어디로 얼마만큼 움직일지에 대한 순간 판단이 몇십 배 빠름}
  \item \intent{VLA는 다른 환경으로 일반화가 안 됨. LLM은 됨. 이걸 엄청 부각하고 항상 기억}
  \item \intent{안전은 크게 다루지 않음}
  \item \intent{관찰: Astra로 로봇을 명령 기반으로 제어하는 프로젝트가 많이 나옴. Astra로 조정하는 방식이 VLA보다 생각보다 성능이 잘 나옴. 하지만 이미지를 보고 엔드포인트를 움직이는 방식에 그침. Astra는 API라서 시간이 너무 오래 걸림}
  \item \intent{파이프라인: 사용자 명령 $\rightarrow$ Astra가 잘게 쪼개 미리 계획 $\rightarrow$ 계획에 맞는 스킬을 불러옴 $\rightarrow$ Astra가 진행 $\rightarrow$ 프레임별로 계획대로 되는지 판별}
  \item \idea{계층 구조, Astra로 로봇 제어~\cite{agp2026,robodojo2026}, 스텝마다 LLM 이산 행동~\cite{showharness2026}은 이미 있음. 안 멈추고 겹쳐 부르는 것까지도 이미 있음~\cite{slowbrain2026}. 우리 자리는 겹친 답을 확정하는 규칙(M4)}
  \item \decide{주장 범위. $\pi_{0.5}$ 등이 새 환경을 주장함 $\rightarrow$ ``대규모 로봇 데이터·학습 없이 덜 무너짐''으로 좁힐지}
  \item \idea{가장 약한 고리: ``같은 술어 위에서 Jev 결정이 룰보다 낫다''를 아무도 안 잼 $\rightarrow$ 첫 실험으로 먼저 잼}
\end{itemize}

\paragraph{컨트리뷰션 후보 (v5)}
\begin{enumerate}
  \item M4: 겹쳐 부른 Jev 답들이 서로 맞는지(a) + 이미 한 동작이 예상대로 됐는지(b) 둘 다 볼 때만 확정 (가장 강함. 비정지·겹침 요청 자체는 선행 있음)
  \item 평가: 같은 인식 앞단 위에서 학습 결정 층 대 LLM 결정 층의 standard$\rightarrow$random 상대 하락폭 (처음 잼)
  \item M8: 비정지 로봇에서 호출 시점 $\times$ effort 평가 (평가 컨트리뷰션. 정체 시간 초과 호출은 이미 있음)
  \item M6: 스킬 내부 결정 지점을 typed 질문으로 (로봇 선점 없음, 비로봇 선례 있음)
  \item M10: Astra가 컴파일한 규칙만 Jev에 (규칙 컴파일 자체는 선행 있음 $\rightarrow$ 좁음)
\end{enumerate}
\chg{v2의 ``호출 방식 비교가 새롭다''는 틀림. 주기 대 이벤트 비교는 이미 여러 편~\cite{checkvla2026,eventtrig2026}}
